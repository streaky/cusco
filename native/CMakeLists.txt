cmake_minimum_required(VERSION 3.25)
project(cusco_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Pinned llama.cpp source")
set(LLAMA_BUILD_DIR "" CACHE PATH "Pinned llama.cpp build")
find_library(LLAMA_LIBRARY llama PATHS "${LLAMA_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
add_library(cusco_executor STATIC shim/cusco_executor.cpp)
target_include_directories(cusco_executor PUBLIC include PRIVATE "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include")
target_link_libraries(cusco_executor PUBLIC "${LLAMA_LIBRARY}")
install(TARGETS cusco_executor ARCHIVE DESTINATION lib)
install(FILES include/cusco_executor.h DESTINATION include)
